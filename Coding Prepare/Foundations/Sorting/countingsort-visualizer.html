def counting_sort(arr):
    max_val = max(arr)
    min_val = min(arr)
    range_size = max_val - min_val + 1
    # Count occurrences
    count = [0] * range_size
    for num in arr:
        count[num - min_val] += 1
    # Cumulative count
    for i in range(1, range_size):
        count[i] += count[i - 1]
    # Build output array
    output = [0] * len(arr)
    for i in range(len(arr) - 1, -1, -1):
        num = arr[i]
        output[count[num - min_val] - 1] = num
        count[num - min_val] -= 1
    return output
# Time: O(n + k) where k=range
# Space: O(n + k)
