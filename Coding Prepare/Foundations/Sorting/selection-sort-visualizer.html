def selection_sort(arr):
    n = len(arr)
    # Outer loop: position to fill
    for i in range(n - 1):
        min_idx = i
        # Find minimum in unsorted part
        for j in range(i + 1, n):
            if arr[j] < arr[min_idx]:
                min_idx = j
        # Swap minimum to position i
        arr[i], arr[min_idx] = arr[min_idx], arr[i]
    return arr
# Time Complexity: O(n²)
# Space Complexity: O(1)
