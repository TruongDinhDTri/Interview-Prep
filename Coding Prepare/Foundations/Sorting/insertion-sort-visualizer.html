def insertion_sort(arr):
    n = len(arr)
    # Start from 2nd element (index 1)
    for i in range(1, n):
        key = arr[i]
        j = i - 1
        # Shift elements right while key is smaller
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        # Insert key at correct position
        arr[j + 1] = key
    return arr
# Time: O(n²) worst, O(n) best
# Space: O(1)
